/** * Weekly Team Performance – individual email reports * --------------------------------------------------- * Every Monday at 9 PM IST, each reviewer gets an email with their * performance for the week that just ended (Mon–Sun). Everyone on the * CC list is copied on every email. * * Setup (one time): see README.md, or run these in order: * 1. setupTeamTab() – creates a "Team" tab listing every member found in the data * 2. Fill in the emails in the Team tab * 3. sendTestToMe() – sends every report to YOU only, so you can check them * 4. installWeeklyTrigger() – turns on the Monday 9 PM schedule */ // ======================= CONFIG ======================= const CONFIG = { DATA_SHEET_GID: 421286318, // the tab that holds the daily work log TEAM_SHEET_NAME: 'Team', // tab with Name | Email | Active CC_EMAILS: [ // the extra people copied on every email // 'person.one@spyne.ai', // 'person.two@spyne.ai', ], DASHBOARD_URL: '', // optional: your GitHub Pages link, shown in the email TIMEZONE: 'Asia/Kolkata', SEND_DAY: ScriptApp.WeekDay.MONDAY, SEND_HOUR: 21, // 9 PM WORKING_DAYS: 6, // Mon–Sat count as expected daily updates SEND_EMPTY_REPORTS: true, // email active members who logged nothing too SUBJECT_PREFIX: 'Weekly Performance Report', }; // ====================================================== /** Main job – run by the weekly trigger. */ function sendWeeklyReports() { run_({ testTo: null }); } /** Sends every member's report to you only (no CC), so you can preview. */ function sendTestToMe() { run_({ testTo: Session.getActiveUser().getEmail() }); } /** Creates/refreshes the Team tab with every name found in the log. Existing emails are kept. */ function setupTeamTab() { const ss = SpreadsheetApp.getActive(); const rows = readLog_(); const names = [...new Set(rows.map(r => r.who))].filter(n => n !== 'Unassigned').sort(); let sh = ss.getSheetByName(CONFIG.TEAM_SHEET_NAME); const existing = {}; if (!sh) { sh = ss.insertSheet(CONFIG.TEAM_SHEET_NAME); } else { sh.getDataRange().getValues().slice(1).forEach(r => { if (r[0]) existing[normName_(r[0]).toLowerCase()] = r; }); } const out = [['Name', 'Email', 'Active (Yes/No)']]; names.forEach(n => { const e = existing[n.toLowerCase()]; out.push([n, e ? e[1] : '', e ? e[2] : 'Yes']); delete existing[n.toLowerCase()]; }); Object.values(existing).forEach(e => out.push([e[0], e[1], e[2] || 'Yes'])); // keep manually added people sh.clear(); sh.getRange(1, 1, out.length, 3).setValues(out); sh.getRange(1, 1, 1, 3).setFontWeight('bold').setBackground('#e8eefc'); sh.setFrozenRows(1); sh.autoResizeColumns(1, 3); SpreadsheetApp.getUi && safeToast_('Team tab ready – add each member\'s email in column B.'); } /** Installs the Monday 9 PM IST trigger (removes any older copy first). */ function installWeeklyTrigger() { ScriptApp.getProjectTriggers() .filter(t => t.getHandlerFunction() === 'sendWeeklyReports') .forEach(t => ScriptApp.deleteTrigger(t)); ScriptApp.newTrigger('sendWeeklyReports') .timeBased() .onWeekDay(CONFIG.SEND_DAY) .atHour(CONFIG.SEND_HOUR) .nearMinute(0) .inTimezone(CONFIG.TIMEZONE) .create(); safeToast_('Weekly emails scheduled: every Monday around 9 PM IST.'); } /** Adds a menu to the sheet so you don't need the script editor. */ function onOpen() { SpreadsheetApp.getUi().createMenu('Weekly Report') .addItem('1. Set up Team tab', 'setupTeamTab') .addItem('2. Send test emails to me', 'sendTestToMe') .addItem('3. Turn on Monday 9 PM schedule', 'installWeeklyTrigger') .addSeparator() .addItem('Send real reports now', 'sendWeeklyReports') .addToUi(); } // ------------------------------------------------------------------ function run_(opts) { const tz = CONFIG.TIMEZONE; const now = new Date(); // Week reported = the last completed Mon–Sun week, relative to today in IST. const todayIST = ymdToDate_(Utilities.formatDate(now, tz, 'yyyy-MM-dd')); const thisMon = mondayOf_(todayIST); const weekStart = addDays_(thisMon, -7); const weekEnd = addDays_(weekStart, 6); const prevStart = addDays_(weekStart, -7); const rows = readLog_(); const team = readTeam_(); if (!team.length) throw new Error('Team tab is empty – run setupTeamTab() and fill in emails.'); const inRange = (r, s) => r.date >= s && r.date <= addDays_(s, 6); const sent = [], skipped = []; team.forEach(m => { if (!m.active) return; if (!m.email) { skipped.push(m.name + ' (no email)'); return; } const mine = rows.filter(r => r.who.toLowerCase() === m.name.toLowerCase()); const cur = mine.filter(r => inRange(r, weekStart)); const prev = mine.filter(r => inRange(r, prevStart)); if (!cur.length && !CONFIG.SEND_EMPTY_REPORTS) { skipped.push(m.name + ' (no entries)'); return; } const html = buildEmail_(m.name, cur, prev, weekStart, weekEnd); const label = fmt_(weekStart, 'd MMM') + ' – ' + fmt_(weekEnd, 'd MMM yyyy'); const subject = CONFIG.SUBJECT_PREFIX + ' | ' + m.name + ' | ' + label; const csv = toCsv_(cur); const mail = { to: opts.testTo || m.email, subject: (opts.testTo ? '[TEST → ' + m.email + '] ' : '') + subject, htmlBody: html, name: 'Team Performance Report', attachments: cur.length ? [Utilities.newBlob(csv, 'text/csv', m.name.replace(/\s+/g, '_') + '_' + Utilities.formatDate(weekStart, tz, 'yyyy-MM-dd') + '.csv')] : [], }; const cc = CONFIG.CC_EMAILS.filter(Boolean).join(','); if (cc && !opts.testTo) mail.cc = cc; MailApp.sendEmail(mail); sent.push(m.name); }); Logger.log('Week %s – %s | sent: %s | skipped: %s', weekStart, weekEnd, sent.join(', '), skipped.join(', ') || 'none'); safeToast_('Sent ' + sent.length + ' report(s)' + (skipped.length ? '. Skipped: ' + skipped.join(', ') : '')); } // ---------------------------- data ---------------------------- function readLog_() { const ss = SpreadsheetApp.getActive(); const sh = ss.getSheets().find(s => s.getSheetId() === CONFIG.DATA_SHEET_GID); if (!sh) throw new Error('Could not find the data tab (gid ' + CONFIG.DATA_SHEET_GID + ').'); const vals = sh.getDataRange().getValues(); const head = vals[0].map(h => String(h).trim().toLowerCase()); const ix = p => head.findIndex(h => h.indexOf(p) === 0); const I = { date: ix('date'), who: ix('work done'), ticket: ix('ticket'), type: ix('review type'), ent: ix('enterprise name'), roof: ix('rooftop name'), link: ix('data/work'), vins: head.indexOf('vins'), imp: ix('impacted'), rem: ix('remarks'), seg: ix('segment'), }; const g = (r, i) => (i < 0 ? '' : r[i]); return vals.slice(1).map(r => { const d = parseDate_(g(r, I.date)); if (!d) return null; return { date: d, who: normName_(g(r, I.who)), ticket: String(g(r, I.ticket)).trim(), type: String(g(r, I.type)).trim() || 'Unspecified', ent: String(g(r, I.ent)).trim() || '—', roof: String(g(r, I.roof)).trim(), link: String(g(r, I.link)).trim(), vins: num_(g(r, I.vins)), imp: num_(g(r, I.imp)), rem: String(g(r, I.rem)).trim(), seg: String(g(r, I.seg)).trim(), }; }).filter(Boolean); } function readTeam_() { const sh = SpreadsheetApp.getActive().getSheetByName(CONFIG.TEAM_SHEET_NAME); if (!sh) return []; return sh.getDataRange().getValues().slice(1) .filter(r => String(r[0]).trim()) .map(r => ({ name: normName_(r[0]), email: String(r[1] || '').trim(), active: !/^(no|n|false|0)$/i.test(String(r[2] || 'Yes').trim()), })); } // ---------------------------- email ---------------------------- function buildEmail_(name, cur, prev, ws, we) { const sum = a => a.reduce((s, r) => s + r, 0); const T = cur.length, V = sum(cur.map(r => r.vins)), I = sum(cur.map(r => r.imp)); const pT = prev.length, pV = sum(prev.map(r => r.vins)), pI = sum(prev.map(r => r.imp)); const days = []; for (let i = 0; i < 7; i++) { const d = addDays_(ws, i); const n = cur.filter(r => sameDay_(r.date, d)).length; days.push({ d, n, working: i < CONFIG.WORKING_DAYS }); } const logged = days.filter(x => x.working && x.n > 0).length; const missed = days.filter(x => x.working && x.n === 0).map(x => fmt_(x.d, 'EEE d MMM')); const byType = {}; cur.forEach(r => { const t = byType[r.type] || (byType[r.type] = { n: 0, v: 0, i: 0 }); t.n++; t.v += r.vins; t.i += r.imp; }); const c = { ink: '#1a1d23', mute: '#667085', line: '#e4e7ec', up: '#12805c', down: '#c4320a', acc: '#2f6fed', soft: '#eef3fe', warn: '#b54708', warnBg: '#fef6ee' }; const chg = (a, b) => { if (!b) return 'no data last week'; const p = Math.round((a - b) / b * 100); const col = p > 0 ? c.up : p < 0 ? c.down : c.mute; return '' + (p > 0 ? Generated automatically from the team work tracker. Impact rate = Impacted VINs ÷ VINs reviewed.

'; return h; } // ---------------------------- helpers ---------------------------- const MON_ = { jan: 0, feb: 1, mar: 2, apr: 3, may: 4, jun: 5, jul: 6, aug: 7, sep: 8, oct: 9, nov: 10, dec: 11 }; function parseDate_(v) { if (v instanceof Date && !isNaN(v)) return ymdToDate_(Utilities.formatDate(v, CONFIG.TIMEZONE, 'yyyy-MM-dd')); const s = String(v || '').trim(); let m = s.match(/^(\d{1,2})[-\/ ]([A-Za-z]{3})[A-Za-z]*[-\/ ](\d{2,4})$/); if (m) { let y = +m[3]; if (y < 100) y += 2000; return new Date(y, MON_[m[2].toLowerCase()], +m[1]); } m = s.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})$/); // dd/mm/yyyy if (m) return new Date(+m[3], +m[2] - 1, +m[1]); return null; } function ymdToDate_(s) { const p = s.split('-'); return new Date(+p[0], +p[1] - 1, +p[2]); } function mondayOf_(d) { const x = new Date(d); x.setDate(x.getDate() - ((x.getDay() + 6) % 7)); x.setHours(0, 0, 0, 0); return x; } function addDays_(d, n) { const x = new Date(d); x.setDate(x.getDate() + n); return x; } function sameDay_(a, b) { return a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate(); } function num_(v) { if (typeof v === 'number') return v; const n = parseFloat(String(v || '').replace(/[^0-9.\-]/g, '')); return isNaN(n) ? 0 : n; } function normName_(s) { s = String(s || '').trim().replace(/\s+/g, ' '); return s ? s.split(' ').map(w => w[0].toUpperCase() + w.slice(1).toLowerCase()).join(' ') : 'Unassigned'; } function fmt_(d, p) { return Utilities.formatDate(d, CONFIG.TIMEZONE, p); } function fmtN_(n) { return Math.round(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ','); } function pct_(a, b) { return b ? (a / b * 100).toFixed(1) + '%' : '–'; } function esc_(s) { return String(s).replace(/[&<>"]/g, ch => ({ '&': '&', '<': '<', '>': '>', '"': '"' }[ch])); } function toCsv_(rows) { const q = v => '"' + String(v).replace(/"/g, '""') + '"'; const head = ['Date', 'Ticket No.', 'Review Type', 'Enterprise', 'Rooftop', 'Link', 'VINs', 'Impacted VINs', 'Remarks', 'Segment']; return [head.map(q).join(',')].concat(rows.map(r => [fmt_(r.date, 'dd-MMM-yy'), r.ticket, r.type, r.ent, r.roof, r.link, r.vins, r.imp, r.rem, r.seg].map(q).join(','))).join('\n'); } function safeToast_(msg) { try { SpreadsheetApp.getActive().toast(msg, 'Weekly Report', 8); } catch (e) { Logger.log(msg); } }
